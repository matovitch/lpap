{\large\bfseries Linear Probing Amplitude Pooling}\\[0.35em]
{\small
$V \in \mathbb{R}^{C \times P}$,
$\Delta \in \mathbb{Z}^{C \times P}$,
$a \in \mathbb{R}^{C}$,
$d \in \mathbb{Z}^{C}$,
$P = N/C$
}

\begin{algorithmic}[1]
\Require $x \in \mathbb{R}^{N}$,\; $C$,\; $K$ \Comment{$N \bmod C = 0$}
\Ensure $a \in \mathbb{R}^{C}$,\; $d \in \mathbb{Z}^{C}$
\State $V \gets \mathrm{reshape}(x,\, C \times P)$
\State $\Delta \gets 0_{C \times P}$;\;
       $a \gets 0_{C}$;\;
       $d \gets 0_{C}$
\For{$k = 0,\ldots,K-1$}
  \State $\ell_i \gets (i - k) \bmod C$
         \Comment{$\forall\, i \in \{0,\ldots,C-1\}$}
  \State $j_i \gets \arg\max_{p}\, |V[\ell_i,\, p]|$
  \State $c_i \gets V[\ell_i,\, j_i]$
  \State $\hat{d}_i \gets \Delta[\ell_i,\, j_i] + k$
  \State $m_i \gets \bigl(|c_i| \ge |a_i|\bigr)$
  \Where{$m$}
    \State $V[\ell,\, j] \gets a$
    \State $\Delta[\ell,\, j] \gets d - k$
    \State $a \gets c$
    \State $d \gets \hat{d}$
  \EndWhere
\EndFor
\State \Return $a$,\; $d$
\end{algorithmic}

\vspace{0.55em}
{\footnotesize
\textbf{Notes.}
\begin{enumerate}
  \itemsep0.25em
  \item The reference implementation also returns $\mathrm{flatten}(V)$, the
        residual work array after in-place swap-backs. It is scratch state,
        not part of the pooled representation $(a,d)$; $\Delta$ likewise.
  \item In the full stack, a fixed seeded \emph{random permutation} is applied
        to $x$ before this procedure and inverted afterward, so structured
        locality in the input does not align with the folded bucket lanes.
\end{enumerate}
}
