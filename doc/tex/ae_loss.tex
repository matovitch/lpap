{\large\bfseries Autoencoder loss}\\[0.45em]
{\footnotesize
Pair losses averaged over parallel widths $C_i$.
}\\[0.7em]
\[
\setlength{\jot}{0.55em}
\begin{aligned}
\mathcal{L}
&=
\frac{1}{|I|}\sum_{i\in I}
\Bigl[
  \lambda_{\mathrm{rec}}\,L_{\mathrm{rec}}^{(i)}
  +
  \lambda_{\mathrm{e}}\,L_{\mathrm{e}}^{(i)}
  +
  \lambda_{\mathrm{ce}}\,L_{\mathrm{ce}}^{(i)}
\Bigr]
\\
L_{\mathrm{rec}}^{(i)}
&=
\lVert x-\hat{x}_i\rVert_2^{2},
\quad
L_{\mathrm{e}}^{(i)}
=
\lVert e-\hat{e}_i\rVert_1,
\quad
L_{\mathrm{ce}}^{(i)}
=
\mathrm{CE}_{i}^{\mathrm{LPAP}}
\end{aligned}
\]
\vspace{0.35em}
{\footnotesize
An optional signed-mass term on the shared latent $e$ exists in config but has
had little practical effect so far; see the training-stack notes.
}
